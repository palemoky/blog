{{- /* 按键速查表，支持合并单元格。数据文件放在文章目录（page bundle）里： {{< keytable "doom-keys.yaml" >}} 数据格式： columns: [分类, 操作组, 按键, 作用] sections: - name: 分类名 groups: # 有 groups 时，分类和操作组两列都会合并 - name: 操作组名 rows: - ['`SPC f f`', 打开文件] - name: 分类名 rows: # 没有 groups 时，只合并分类这一列 - [打开文件, '`SPC f f`', '`ff`'] 单元格内容按 Markdown 渲染，按键用反引号包起来。 */ -}} {{- $file := .Get 0 -}} {{- $res := .Page.Resources.Get $file -}} {{- if not $res -}} {{- errorf "keytable: %s 不在 %s 的目录里" $file .Page.File.Path -}} {{- end -}} {{- $data := $res | transform.Unmarshal -}} {{- $grouped := false -}} {{- range $data.sections }}{{ if .groups }}{{ $grouped = true }}{{ end }}{{ end -}}
{{- range $data.columns }}{{ end -}} {{- range $data.sections }} {{- $section := . -}} {{- $groups := slice -}} {{- if .groups -}} {{- $groups = .groups -}} {{- else -}} {{- $groups = slice (dict "rows" .rows) -}} {{- end -}} {{- $total := 0 -}} {{- range $groups }}{{ $total = add $total (len .rows) }}{{ end -}} {{- range $gi, $group := $groups }} {{- range $ri, $row := $group.rows }} {{- if and (eq $gi 0) (eq $ri 0) }} {{- end }} {{- if and $section.groups (eq $ri 0) }} {{- end }} {{- range $row }} {{- end }} {{- end }} {{- end }} {{- end }}
